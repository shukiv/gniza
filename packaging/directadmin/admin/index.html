#!/bin/sh # Gniza's administrator page on DirectAdmin. # # DirectAdmin runs this as the account named in plugin.conf's # admin_run_as, and hands it the caller's panel session as SESSION_ID and # SESSION_KEY. This forwards the request to Gniza over that account's own # socket with the session on it, and Gniza puts the session to # DirectAdmin before it serves anything: the account this runs as is a # service account, so it proves nothing about who is at the keyboard, and # the session is the whole of the authorization. See ADR 0020. # # The output is the page body. DirectAdmin's skin supplies the document # around it, which is why nothing here writes CGI headers. set -eu umask 077 PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin export PATH SOCKET=/var/run/gniza/directadmin/plugin.sock METHOD=${REQUEST_METHOD:-GET} # DirectAdmin sets QUERY_STRING from the address bar, and a relative # link from this page keeps the plugin's own path -- measured on 1.709, # both at /CMD_PLUGINS_ADMIN/gniza/index.html and inside Evolution's # wrapper at /evo/plugin?src=..., where ?p=accounts arrived as # QUERY_STRING=p=accounts. That is what Gniza's routing is built on. QUERY=${QUERY_STRING:-} fail() { # The message is the whole page, so it says what to do rather than # what went wrong internally. Nothing about the session appears in it. printf '

Gniza

%s

\n' "$1" exit 0 } command -v curl >/dev/null 2>&1 || fail 'curl is not installed on this server, and this page reaches the backup service through it.' [ -S "$SOCKET" ] || fail 'The Gniza service is not running. Start it with systemctl start gniza and reload this page.' # The session values travel as headers rather than cookies: nothing here # is a browser, and the only thing that may set them is this script. # Whatever arrives in them is a claim until DirectAdmin confirms it. # # What Gniza answers is passed through as it is, refusals included, so an # operator reads Gniza's own words about why a page did not open. Only a # request that never reached it falls through to the message below. set -- \ --silent --max-time 300 \ --unix-socket "$SOCKET" \ --header "X-Gniza-DA-Session: ${SESSION_ID:-}" \ --header "X-Gniza-DA-Key: ${SESSION_KEY:-}" if [ "$METHOD" = POST ]; then # DirectAdmin does not hand a plugin the request body. It reads the # body itself and puts the whole url-encoded form in POST, leaving # CONTENT_LENGTH empty and stdin at end of file -- measured on 1.709, # where a form of two fields arrived as # POST=gniza_probe_field=...&csrf=... and stdin held nothing. So the # form is forwarded from there, and reading stdin would forward an # empty body to every save. set -- "$@" \ --request POST \ --header "Content-Type: application/x-www-form-urlencoded" \ --data-raw "${POST:-}" fi curl "$@" "http://gniza/?$QUERY" 2>/dev/null && exit 0 fail 'The backup service did not answer. Check systemctl status gniza.'
